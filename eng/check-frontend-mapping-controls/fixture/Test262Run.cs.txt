namespace Fixture;

internal static class Test262Run
{
    private static Test262Outcome Refused(string relativePath, string variant, IReadOnlyList<SliceSourceDiagnostic> diagnostics)
    {

        if (first is not null && first.Code is SliceSourceDiagnosticCode.ConstructOutsideManifest
                or SliceSourceDiagnosticCode.BigIntWiderThanTheFormat)

    }
}
