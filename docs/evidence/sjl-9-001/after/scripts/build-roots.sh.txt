#!/usr/bin/env bash
# build-roots.sh <base|after>: the slice compiler, the command-line root and the conformance root, built in Release
# with --disable-build-servers, one at a time, each behind the gate, in the clean git archive of the commit on tmpfs.
S=/tmp/claude-0/-home-user/0a6f6b61-1da7-5ce6-9020-88003835d4eb/scratchpad/sjl-rest/vm-sjl96
side=$1; T=/dev/shm/sjl96/$side; O=$S/$side/builds; mkdir -p $O
for p in SliceCompiler Cli Conformance; do
  proj=src/compositions/Broiler.VM.Composition.JavaScript.$p/Broiler.VM.Composition.JavaScript.$p.csproj
  $S/scripts/run.sh build-$p-$side $O $T -- dotnet build $proj -c Release --disable-build-servers -nologo || { echo "build $p failed"; exit 4; }
  grep -E 'warning [A-Z]+[0-9]+' $O/build-$p-$side.out.txt | sort -u > $O/build-$p-$side.warnings.txt
done
echo DONE > $O/DONE
