"""Part A of bundle sjl-9-001 (SJL-9.6): completeness of the four hosted runs, every changed verdict of the two
comparisons with its class, and the table of C1 rows, from files the bundle retains. Reads files and git; runs nothing.
Bundle sjl-13-001's part-a-tables.py (task 7.3), with this rule's classes.

usage: part-a-tables.py <bundle> <repo> <parta-dir> [<rerun-dir> [<race-dir>]]

The comparison (eng/compare-test262-forms.py --same-form) ignores a (path, variant) present in one report only and
says how many it ignored. A file whose features include canonical-tz is skipped at the base as one row with no variant
(`-`), the proposal not admitted, and scored after in its variants, so those rows are in one report each: they are
read here from the reports themselves, by path, and listed beside the comparison's rows. Every such row is classified
too; a path present in one report only that is not such a file is UNCLASSIFIED.

Classes, as the rule names them:
  C1 (Z8, kind 1)  a variant of a file whose features include canonical-tz, skipped at the base because the proposal was
                   not admitted, and passing after;
  C1 (Z8, kind 2)  a variant failing at the base that passes after because the formatter reports the identifier as
                   given: read here only for a file that exercises row Z8 in the correspondence's intl402-map.tsv, and
                   listed with that reading for the README to judge;
  C4 (a), (b)      the comparison's own classes, native pair only;
  C4 (c')          the comparison's class (c'), with its re-runs: admitted when every re-run at both commits gives one
                   verdict and dimension; else (e);
  C4 (e)           the race: admitted when the two runs at ten times the wall clock give the same verdict, neither a
                   wall-clock exhaustion nor an internal defect;
  everything else  REGRESSION.
"""
import collections, gzip, json, re, subprocess, sys, pathlib
B, REPO, OUT = pathlib.Path(sys.argv[1]), sys.argv[2], pathlib.Path(sys.argv[3])
RERUN = pathlib.Path(sys.argv[4]) if len(sys.argv) > 4 else None
RACE = pathlib.Path(sys.argv[5]) if len(sys.argv) > 5 else None
BASE, AFTER = '3b33530d354431675667087a0044648b8f617c2e', '68cad8baf7403125d44d5007441d30ff07bc4edf'
FORMS = ('wide-bytecode', 'wide-native')


def git(*a):
    return subprocess.run(['git', '-C', REPO] + list(a), capture_output=True, text=True, check=True).stdout.strip()


lines = ['# part A, SJL-9.6: completeness of each hosted run as the rule states it (bundle sjl-6-001\'s reading), read from the imported artifacts']
for path in ('.github/workflows/test262-evidence.yml', 'eng/test262-evidence.py'):
    lines.append('%s: blob %s at the base commit, %s at the after commit, %s at 0ffc037f (the base run\'s dispatched ref), %s at 11c53897 (the after run\'s)' % (
        path, git('rev-parse', BASE + ':' + path), git('rev-parse', AFTER + ':' + path),
        git('rev-parse', '0ffc037ff0c3fb1a70e090bcfefd66a885f52eee:' + path), git('rev-parse', '11c53897dc7f19e5ef195aa584df0aaabe4ab2c5:' + path)))
reports = {}
for side in ('base', 'after'):
    h = B / side / 'hosted'
    imported = (h / 'imported.txt').read_text().splitlines()
    run_txt = (h / 'run.txt').read_text()
    lines.append('## %s (%s)' % (side, [l for l in imported if l.startswith('workflow run:')][0]))
    lines.append('run.txt: ' + '; '.join(l for l in run_txt.splitlines() if l.split(':')[0] in ('dispatched ref', 'tested commit', 'tested tree', 'inputs')))
    for form in FORMS:
        shards = sorted((h / 'shards' / form).glob('shard-*.log.gz'))
        merge = (h / (form + '.merge.log')).read_text()
        m = re.search(r'merged (\d+) test262 shard report', merge)
        run = (h / (form + '.run.log')).read_text()
        ec = re.findall(r'^exit code: (\d+)', run, re.M)
        j = json.load(open(h / (form + '.test262.json')))
        binl = (h / (form + '.binaries.log')).read_text()
        dl = re.search(r'distinct lists: (\d+)', binl)
        lines.append('%s: shard logs %d of %d; merge took %s shard reports; run log exit code %s; retention retainable=%s; coverage whole=%s; every variant accounted=%s; distinct assembly lists %s' % (
            form, len(shards), j['partition']['shardCount'], m.group(1) if m else '?', ec[-1] if ec else '?', j['retention']['retainable'],
            j['coverage']['whole'], j['totals']['accountsForEveryVariant'], dl.group(1) if dl else '?'))
        rows = {}
        for l in gzip.open(h / (form + '.report.gz'), 'rt'):
            if l.startswith('result|'):
                c = l.rstrip('\n').split('|')
                rows[(c[1], c[2])] = c
        reports[(side, form)] = rows
cmp = {}
for form in FORMS:
    t = (OUT / ('compare-%s.log' % form)).read_text()
    cmp[form] = t
    m = re.search(r'(\d+) variant\(s\) present in only one report were ignored', t)
    lines.append('the comparison %s: variants present in only one report, ignored by it: %s; its exit: %s' % (form, m.group(1) if m else '?', re.findall(r'^exit=(\d+)', t, re.M)))
(OUT / 'completeness.txt').write_text('\n'.join(lines) + '\n')

# Z8 files of the correspondence (intl402-map.tsv: path, rows, ...)
z8_files = set()
for l in open(B / 'intl402-map.tsv', encoding='utf-8'):
    if l.startswith('#') or '\t' not in l:
        continue
    c = l.split('\t')
    if re.search(r'\bZ8\b', c[1]):
        z8_files.add(c[0])


def verdict(c):
    return '%s/%s' % (c[3], c[5]) if c[3] == 'Exhausted' and c[5] else c[3]


rer = collections.defaultdict(list)
if RERUN:
    for form in FORMS:
        p = RERUN / ('rerun-%s.tsv' % form)
        if p.exists():
            for l in open(p):
                side, f2, rep, res = l.rstrip('\n').split('\t', 3)
                c = res.split('|')
                rer[(form, c[1], c[2], side)].append(verdict(c))
race = {}
if RACE:
    for form in FORMS:
        p = RACE / ('race-%s.tsv' % form)
        if p.exists():
            for l in open(p):
                side, f2, rep, res = l.rstrip('\n').split('\t', 3)
                c = res.split('|')
                race[(form, c[1], c[2], side)] = verdict(c)

out_rows, c1_rows = [], []
for form in FORMS:
    cls = None
    for l in cmp[form].splitlines():
        h = re.match(r"## \((.+?)\)", l)
        if h:
            cls = h.group(1)
            continue
        m = re.match(r"  (test/\S+) \[(\w+)\]: (\S+) -> (\S+)", l)
        if not m:
            continue
        path, var, b, a = m.groups()
        if cls.startswith('unclassified'):
            if b == 'Failed' and a == 'Passed' and path in z8_files:
                kind, reading = 'C1 (Z8, kind 2) - to be read', 'fails at the base and passes after; the file exercises row Z8 in intl402-map.tsv'
                c1_rows.append((form, path, var, b, a, 'Z8', 'kind 2: a variant failing at the base that passes after because the formatter reports the identifier as given'))
            else:
                kind, reading = 'REGRESSION', 'a difference in no class the rule admits'
            out_rows.append((form, path, var, b, a, 'unclassified', kind, '-', '-', reading))
        elif cls.startswith("c'"):
            rb, ra = rer.get((form, path, var, 'base'), []), rer.get((form, path, var, 'after'), [])
            if not rb or not ra:
                kind, reading = "C4 (c') - re-runs not yet taken", '-'
            elif len(set(rb + ra)) == 1:
                kind, reading = "C4 (c')", 're-runs agree'
            else:
                rc = (race.get((form, path, var, 'base')), race.get((form, path, var, 'after')))
                if rc[0] is None or rc[1] is None:
                    kind, reading = 'C4 (e) - race not yet taken', 're-runs differ'
                else:
                    okr = rc[0] == rc[1] and not rc[0].endswith('WallClock') and 'Defect' not in rc[0]
                    kind = 'C4 (e)' if okr else 'REGRESSION'
                    reading = 're-runs differ; race at ten times the wall clock: base %s, after %s' % rc
            out_rows.append((form, path, var, b, a, "(c')", kind, ', '.join(rb) or '-', ', '.join(ra) or '-', reading))
        elif cls.startswith('a)') or cls.startswith('b)') or cls in ('a', 'b') or cls.startswith('a ') or cls.startswith('b '):
            ok = form == 'wide-native'
            out_rows.append((form, path, var, b, a, '(%s)' % cls[0], 'C4 (%s)' % cls[0] if ok else 'REGRESSION', '-', '-', 'the comparison\'s own class, both runs native' if ok else 'a native class in the bytecode pair'))
        else:
            out_rows.append((form, path, var, b, a, cls, 'REGRESSION', '-', '-', 'a class the rule does not admit'))
    # rows in one report only
    rb, ra = reports[('base', form)], reports[('after', form)]
    only_b = sorted(k for k in rb if k not in ra)
    only_a = sorted(k for k in ra if k not in rb)
    base_skipped_tz = {k[0] for k in only_b if rb[k][3] == 'Skipped' and 'canonical-tz' in rb[k][7].split(',') and k[1] == '-'}
    for k in only_b:
        if k[0] in base_skipped_tz and any(x[0] == k[0] for x in only_a):
            continue
        out_rows.append((form, k[0], k[1], verdict(rb[k]), '(absent)', 'in the base report only', 'REGRESSION', '-', '-', 'a row the after report does not hold'))
    for k in only_a:
        c = ra[k]
        if k[0] in base_skipped_tz and 'canonical-tz' in c[7].split(','):
            if c[3] == 'Passed':
                kind, reading = 'C1 (Z8, kind 1)', 'canonical-tz, skipped at the base (%s) and passing after' % rb[(k[0], '-')][8]
                c1_rows.append((form, k[0], k[1], 'Skipped', 'Passed', 'Z8', 'kind 1: a variant of a file whose features include canonical-tz, skipped at the base because the proposal was not admitted, and passing after'))
            else:
                kind, reading = 'REGRESSION', 'a canonical-tz variant scored after that does not pass'
            out_rows.append((form, k[0], k[1], 'Skipped (one row, variant -)', verdict(c), 'in the after report only', kind, '-', '-', reading))
        else:
            out_rows.append((form, k[0], k[1], '(absent)', verdict(c), 'in the after report only', 'REGRESSION', '-', '-', 'a row the base report does not hold'))
    # canonical-tz files still skipped after
    for k, c in ra.items():
        if c[3] == 'Skipped' and 'canonical-tz' in c[7].split(','):
            out_rows.append((form, k[0], k[1], verdict(rb.get(k, c)), 'Skipped', 'unchanged', 'REGRESSION', '-', '-', 'a canonical-tz file still skipped after'))

with open(OUT / 'changed-verdicts.tsv', 'w', newline='\n') as o:
    o.write('# part A, SJL-9.6: every variant whose verdict or exhausted dimension differs between the base and the after run, by form, with its class: the comparison\'s rows, and the rows in one report only, which it ignores (part-a-tables.py)\n')
    o.write('form\tpath\tvariant\tbase (hosted)\tafter (hosted)\tcomparison\'s class\tclass and kind\tre-runs at the base\tre-runs at the after commit\treading\n')
    for r in out_rows:
        o.write('\t'.join(r) + '\n')
with open(OUT / 'c1-table.tsv', 'w', newline='\n') as o:
    o.write('# part A, SJL-9.6: the table of C1 rows - path, variant, row and the clause each follows (the rule\'s class C1); bytecode pair then native pair\n')
    o.write('form\tpath\tvariant\tbase (hosted)\tafter (hosted)\trow\tclause\n')
    for r in c1_rows:
        o.write('\t'.join(r) + '\n')
cnt = collections.Counter((r[0], r[6]) for r in out_rows)
print(open(OUT / 'completeness.txt').read())
for k, v in sorted(cnt.items()):
    print('%s\t%s\t%d' % (k[0], k[1], v))
