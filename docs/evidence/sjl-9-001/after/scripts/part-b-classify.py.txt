"""Part B of bundle sjl-9-001 (SJL-9.6): every difference between the base and the after commit under src/tests/cldr/
and src/tests/temporal/, read with git alone, each line classified against classes Z8 and L18 of the rule as its
revision of 2026-11-04 reads them, with the line at both commits and the retained answer's line. Runs nothing.

usage: part-b-classify.py <repo> <base> <after> <out.tsv> <summary.txt>

A data line of a divergences file is "<key> => <answer>"; it replaces the retained answer's line whose text before
` => ` is the same. So a key's answer at a commit is its divergence line there if it has one, else the retained
answer's line (the check passes at both commits, so this is what the program printed). Each changed data line is
read by its key:
  - Z8 (src/tests/cldr/*/ or src/tests/temporal/**/divergences.txt): the key is a time-zone identifier; its answer
    at the base names another identifier (the base reports a zone's primary, decision D15, so that name is the
    key's primary and the key is not its own); after, the answer names the key itself; the text after the name
    (from " | ") is the same at both commits; and the line is rewritten (a divergence line at both), removed (the
    retained line already names the key itself) or added (the retained line names another name).
  - L18 (src/tests/cldr/locales/divergences.txt alone): the key asks `collations` of a locale with no -u-co- keyword
    and no "collation" option, whose language (the options' "language" when it gives one, else the tag's first
    subtag) is neither und nor de nor en; its answer changes only from [emoji,eor] to []; and the line is rewritten
    (a divergence line holding [emoji,eor] at the base), added (the retained line reads [emoji,eor]) or removed (the
    retained line reads []).
A comment line is read as the header of the group whose data lines follow it in its file at the commit that holds
it: a Z8 header heads at least one Z8 line, an L18 header at least one L18 line. Anything else is OTHER, which the
rule makes a regression.
"""
import json, re, subprocess, sys

REPO, BASE, AFTER, OUT, SUMMARY = sys.argv[1:6]
FOLDERS = ['src/tests/cldr', 'src/tests/temporal']


def git(*args):
    return subprocess.run(['git', '-C', REPO] + list(args), capture_output=True, text=True, check=True).stdout


def show(commit, path):
    try:
        return git('show', '%s:%s' % (commit, path))
    except subprocess.CalledProcessError:
        return None


def retained_of(commit, divergences):
    folder = divergences.rsplit('/', 1)[0]
    names = [n for n in git('ls-tree', '--name-only', commit + ':' + folder).split() if n.endswith('.txt') and n != 'divergences.txt']
    return [folder + '/' + n for n in names]


def keyed(text):
    out = {}
    for line in text.splitlines():
        if line.startswith('#') or ' => ' not in line:
            continue
        out.setdefault(line.split(' => ', 1)[0], []).append(line)
    return out


names = [n for n in git('diff', '--name-only', BASE, AFTER, '--', *FOLDERS).split() if n]
rows, other = [], 0
summary = ['# part B, SJL-9.6: every difference between the base %s and the after commit %s under %s, by `git diff`' % (BASE, AFTER, ' and '.join(FOLDERS)),
           '# changed paths: %d' % len(names)] + ['#   ' + n for n in names]
for path in names:
    if not path.endswith('divergences.txt'):
        rows.append((path, '-', '-', 'OTHER', 'a file other than a divergences file changed', '-', '-', '-'))
        other += 1
        continue
    btext, atext = show(BASE, path) or '', show(AFTER, path) or ''
    retained_paths = sorted(set(retained_of(BASE, path)) | set(retained_of(AFTER, path)))
    rb = {p: show(BASE, p) for p in retained_paths}
    ra = {p: show(AFTER, p) for p in retained_paths}
    if rb != ra:
        rows.append((path, '-', '-', 'OTHER', 'a retained answer beside it changed', '-', '-', '-'))
        other += 1
    retained = {}
    for p in retained_paths:
        for k, v in keyed(rb[p] or '').items():
            retained.setdefault(k, []).extend(v)
    bk, ak = keyed(btext), keyed(atext)
    # the diff, line by line (no context), with each line's number at its side
    diff = git('diff', '-U0', BASE, AFTER, '--', path)
    removed, added = [], []
    bl = al = 0
    for line in diff.splitlines():
        m = re.match(r'@@ -(\d+)(?:,\d+)? \+(\d+)(?:,\d+)? @@', line)
        if m:
            bl, al = int(m.group(1)), int(m.group(2))
            continue
        if line.startswith('---') or line.startswith('+++'):
            continue
        if line.startswith('-'):
            removed.append((bl, line[1:])); bl += 1
        elif line.startswith('+'):
            added.append((al, line[1:])); al += 1
    changed_keys = []
    for _, text in removed + added:
        if not text.startswith('#') and ' => ' in text:
            k = text.split(' => ', 1)[0]
            if k not in changed_keys:
                changed_keys.append(k)
    classified = {}
    for k in changed_keys:
        b, a, r = bk.get(k), ak.get(k), retained.get(k)
        beff, aeff = (b or r), (a or r)
        how = 'rewritten' if b and a else 'removed' if b else 'added' if a else '?'
        bline = b[0] if b and len(b) == 1 else (' || '.join(b) if b else '(none)')
        aline = a[0] if a and len(a) == 1 else (' || '.join(a) if a else '(none)')
        rline = r[0] if r and len(r) == 1 else (' || '.join(r) if r else '(none)')
        cls, why = 'OTHER', ''
        single = all(x is None or len(x) == 1 for x in (b, a, r)) and beff and aeff
        if not single:
            why = 'the key has no single line at one side'
        elif path.startswith('src/tests/cldr/locales/'):
            m = re.match(r'^(\d+) (\S+)(?: (\{.*\}))? collations$', k)
            bans, aans = beff[0].split(' => ', 1)[1], aeff[0].split(' => ', 1)[1]
            if m:
                tag, opts = m.group(2), json.loads(m.group(3)) if m.group(3) else {}
                lang = (opts.get('language') or tag.split('-')[0]).lower()
                co = '-u-' in tag.lower() and re.search(r'-co-', tag.lower()) is not None
                ok_how = ((how == 'rewritten' and b[0].endswith(' => [emoji,eor]'))
                          or (how == 'added' and r and r[0].endswith(' => [emoji,eor]'))
                          or (how == 'removed' and r and r[0].endswith(' => []')))
                if (not co and 'collation' not in opts and lang not in ('und', 'de', 'en')
                        and bans == '[emoji,eor]' and aans == '[]' and ok_how):
                    cls, why = 'L18', 'collations of language %s, which the Collator lacks: [emoji,eor] to [], %s' % (lang, how)
                else:
                    why = 'collations line outside L18 (language %s, co keyword %s, collation option %s, %s to %s, %s)' % (lang, co, 'collation' in opts, bans, aans, how)
            else:
                why = 'a locales line that asks no collations'
        else:
            # Z8: "<id> => <name> | <rest>"
            def split(line):
                rhs = line.split(' => ', 1)[1]
                name, _, rest = rhs.partition(' | ')
                return name, rest
            (bn, brest), (an, arest) = split(beff[0]), split(aeff[0])
            ok_how = (how == 'rewritten' or (how == 'removed' and r and split(r[0])[0] == k)
                      or (how == 'added' and r and split(r[0])[0] != k))
            if '/' in k or re.match(r'^[A-Za-z][A-Za-z0-9+_-]*$', k):
                if bn != k and an == k and brest == arest and ok_how:
                    cls, why = 'Z8', 'resolved name moves from the primary %s to the identifier as given, %s' % (bn, how)
                else:
                    why = 'zone line outside Z8 (base name %s, after name %s, rest same %s, %s)' % (bn, an, brest == arest, how)
            else:
                why = 'not a time-zone identifier key'
        classified[k] = cls
        if cls == 'OTHER':
            other += 1
        rows.append((path, 'data', k, cls, why, bline, aline, rline))
    # comments: each changed comment line, read as the header of the group that follows it
    for side, lines_, text, keys in (('base', removed, btext, bk), ('after', added, atext, ak)):
        all_lines = text.splitlines()
        for number, line in lines_:
            if not line.startswith('#') and line.strip():
                continue
            if not line.strip():
                cls = 'blank'
                why = 'a blank line between groups'
            else:
                heads = []
                for follow in all_lines[number:]:
                    if follow.startswith('#') or not follow.strip():
                        if heads and not follow.strip():
                            break
                        continue
                    heads.append(follow.split(' => ', 1)[0])
                kinds = sorted({classified.get(h) for h in heads if classified.get(h)})
                cls = ('header of ' + '/'.join(kinds)) if kinds and 'OTHER' not in kinds else 'OTHER'
                why = 'comment line %d at the %s, heading %d lines, %d of them changed (%s)' % (number, side, len(heads), sum(1 for h in heads if h in classified), ', '.join(kinds) or 'none')
            if cls == 'OTHER':
                other += 1
            rows.append((path, 'comment', '%s:%d' % (side, number), cls, why, line if side == 'base' else '-', line if side == 'after' else '-', '-'))
with open(OUT, 'w', newline='\n') as o:
    o.write('# part B, SJL-9.6: every changed line under src/tests/cldr/ and src/tests/temporal/ between %s (base) and %s (after), with its class; for a data line, its line at the base, after, and the retained answer\'s line (part-b-classify.py)\n' % (BASE, AFTER))
    o.write('path\tkind\tkey\tclass\treading\tline at the base\tline after\tretained answer\'s line\n')
    for row in rows:
        o.write('\t'.join(row) + '\n')
from collections import Counter
c = Counter((r[0], r[1], r[3]) for r in rows)
summary.append('# lines by path, kind and class:')
for (p, kind, cls), n in sorted(c.items()):
    summary.append('%s\t%s\t%s\t%d' % (p, kind, cls, n))
hows = Counter((r[0], r[3], r[4].rsplit(', ', 1)[-1]) for r in rows if r[1] == 'data')
summary.append('# data lines by path, class and how the change made them:')
for (p, cls, how), n in sorted(hows.items()):
    summary.append('%s\t%s\t%s\t%d' % (p, cls, how, n))
summary.append('# lines outside Z8 and L18 (OTHER): %d' % other)
open(SUMMARY, 'w', newline='\n').write('\n'.join(summary) + '\n')
print('\n'.join(summary))
