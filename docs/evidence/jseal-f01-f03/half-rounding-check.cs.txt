using System;
// Reference: exact binary16 rounding from double, roundTiesToEven, via integer arithmetic.
static ushort Ref(double d)
{
    ulong bits = (ulong)BitConverter.DoubleToInt64Bits(d);
    ushort sign = (ushort)((bits >> 48) & 0x8000);
    if (double.IsNaN(d)) return (ushort)(sign | 0x7E00);
    double a = Math.Abs(d);
    if (double.IsInfinity(a)) return (ushort)(sign | 0x7C00);
    // scale: value = a; binary16 min subnormal 2^-24. Compute q = a / 2^-24 for subnormal range, otherwise per exponent.
    int e = Math.ILogB(a); // floor(log2 a)
    if (a == 0) return sign;
    int qexp = Math.Max(e, -14) - 10; // ulp exponent
    double scaled = Math.ScaleB(a, -qexp); // exact (power of two scaling), in [0, 2048)
    double fl = Math.Floor(scaled);
    double frac = scaled - fl; // exact
    ulong m = (ulong)fl;
    if (frac > 0.5 || (frac == 0.5 && (m & 1) == 1)) m++;
    long enc = ((long)(qexp + 25) << 10) + (long)m - 1024;
    if (enc >= 0x7C00) return (ushort)(sign | 0x7C00);
    return (ushort)(sign | (int)enc);
}

void Check(double d) {
    ushort want = Ref(d); ushort got = BitConverter.HalfToUInt16Bits((Half)d);
    if (double.IsNaN(d)) { if (!Half.IsNaN((Half)d)) bad++; return; }
    if (want != got) { if (bad++ < 10) Console.WriteLine($"MISMATCH {d:R} bits {BitConverter.DoubleToInt64Bits(d):X16} want {want:X4} got {got:X4}"); }
}
