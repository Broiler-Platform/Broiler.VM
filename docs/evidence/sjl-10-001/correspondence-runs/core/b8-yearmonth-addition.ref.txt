This polyfill should only be used to run tests or to experiment in the browser devtools console.
To polyfill Temporal in your own projects, see https://github.com/tc39/proposal-temporal#polyfills.
PlainYearMonth("2020-01").subtract({months: 1}) => Temporal.PlainYearMonth 2019-12
PlainYearMonth("2020-01").add({years: 1, months: 13}) => Temporal.PlainYearMonth 2022-02
PlainYearMonth("2020-01").add({days: 1}) => THROWS RangeError: only years and months can be added to Temporal.PlainYearMonth
PlainYearMonth("2020-01").add({weeks: 1}) => THROWS RangeError: only years and months can be added to Temporal.PlainYearMonth
PlainYearMonth("2020-01").add({hours: 1}) => THROWS RangeError: only years and months can be added to Temporal.PlainYearMonth
PlainYearMonth("2020-01").add({days: 0, months: 1}) => Temporal.PlainYearMonth 2020-02
PlainYearMonth("2020-01").add({months: 1}, {overflow: "reject"}) => Temporal.PlainYearMonth 2020-02
PlainYearMonth("2020-01").add({months: -1200000}) => Temporal.PlainYearMonth -097980-01
exit=0
