#!/usr/bin/env bash
# checks.sh <commit> <name>: SJL-13's exit reading (Broiler.VM). A clean git archive of <commit> on /dev/shm, a Release build of
# Broiler.VM.slnx (warnings listed), the architecture tests (with a TRX of every test) and the contract tests, each through
# the flock'd SCRATCH/sjl-rest/s2/ev2/scripts/run.sh (lock /dev/shm/s2-gate.lock, hard idle gate). Kills nothing of others.
E=/tmp/claude-0/-home-user/0a6f6b61-1da7-5ce6-9020-88003835d4eb/scratchpad/sjl-rest/exit13
R=/tmp/claude-0/-home-user/0a6f6b61-1da7-5ce6-9020-88003835d4eb/scratchpad/sjl-rest/s2/ev2/scripts/run.sh
REPO=/home/user/vm-s2ev
c=$(git -C $REPO rev-parse "$1"); n=$2; T=/dev/shm/exit13/$n; O=$E/$n
[ -e "$T" ] && { echo "REFUSED: $T exists"; exit 3; }
mkdir -p "$T" "$O"
git -C $REPO archive --format=tar "$c" | tar -x -C "$T"
{ echo "commit $c"; echo "tree $(git -C $REPO rev-parse $c^{tree})"; echo "archived $(date -u +%FT%TZ) into $T"; echo "files $(find $T -type f | wc -l)"; } > $O/archive.txt
$R build-solution $O $T -- dotnet build Broiler.VM.slnx -c Release --disable-build-servers -nologo; echo "build rc=$?" >> $O/status.txt
grep -E 'warning [A-Z]+[0-9]+' $O/build-solution.out.txt | sort -u > $O/build-warnings.txt
grep -E '^ *[0-9]+ Warning\(s\)|^ *[0-9]+ Error\(s\)|Build succeeded|Time Elapsed' $O/build-solution.out.txt | tail -4 > $O/build-summary.txt
$R test-architecture $O $T -- dotnet test src/tests/Broiler.VM.Architecture.Tests/Broiler.VM.Architecture.Tests.csproj -c Release --no-build --disable-build-servers -nologo --logger "trx;LogFileName=architecture.trx" --results-directory $O/trx; echo "arch rc=$?" >> $O/status.txt
$R test-contract $O $T -- dotnet test src/tests/Broiler.VM.Contract.Tests/Broiler.VM.Contract.Tests.csproj -c Release --no-build --disable-build-servers -nologo --logger "trx;LogFileName=contract.trx" --results-directory $O/trx; echo "contract rc=$?" >> $O/status.txt
python3 -I $E/scripts/rules-from-trx.py $O/trx/architecture.trx $c > $O/rules.tsv 2>> $O/status.txt
rm -rf "$T"; rmdir /dev/shm/exit13 2>/dev/null
echo DONE > $O/DONE
