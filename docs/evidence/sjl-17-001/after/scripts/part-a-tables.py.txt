"""Part A of bundle sjl-17-001 (SJL-17.6): completeness of the four hosted runs, every changed verdict of the two
comparisons with its class, the rows in one report only, and the collation directories' rows, from files the bundle
retains. Reads files and git; runs nothing. Bundle sjl-9-001's part-a-tables.py, with this rule's classes.

usage: part-a-tables.py <bundle> <repo> <parta-dir> [<rerun-dir> [<race-dir>]]

Classes, as the rule names them:
  C1               a variant failing at the base that passes after; JSD-0084 names none, so each is listed for the
                   README to read against "because the shared collator follows the pinned text", and is a finding;
  C4 (a), (b)      the comparison's own classes, native pair only ((b) only under --exempt-guest-loads);
  C4 (c')          the comparison's class (c'), with its re-runs: admitted when every re-run at both commits gives one
                   verdict and dimension (section 1's reading 6); else (e);
  C4 (e)           the race: admitted when the two runs at ten times the wall clock give the same verdict, neither a
                   wall-clock exhaustion nor an internal defect;
  everything else  REGRESSION, a row in one report only among them (section 1's reading 7).
"""
import collections, gzip, json, re, subprocess, sys, pathlib
B, REPO, OUT = pathlib.Path(sys.argv[1]), sys.argv[2], pathlib.Path(sys.argv[3])
RERUN = pathlib.Path(sys.argv[4]) if len(sys.argv) > 4 else None
RACE = pathlib.Path(sys.argv[5]) if len(sys.argv) > 5 else None
BASE, AFTER = '68cad8baf7403125d44d5007441d30ff07bc4edf', '983d690adac63f6d117ff2cd2d9cca61fc16e3d2'
FORMS = ('wide-bytecode', 'wide-native')
COLLATION = ('test/intl402/Collator/', 'test/built-ins/String/prototype/localeCompare/', 'test/intl402/String/prototype/localeCompare/')


def git(*a):
    return subprocess.run(['git', '-C', REPO] + list(a), capture_output=True, text=True, check=True).stdout.strip()


lines = ['# part A, SJL-17.6: completeness of each hosted run as the rule states it (bundle sjl-6-001\'s reading), read from the imported artifacts']
for path in ('.github/workflows/test262-evidence.yml', 'eng/test262-evidence.py', 'eng/run-test262.py', 'eng/collect-js-release-candidate.py', 'eng/compare-test262-forms.py', 'src/tests/conformance/pins/test262.pin'):
    lines.append('%s: blob %s at the base commit, %s at the after commit (both runs\' dispatched ref)' % (
        path, git('rev-parse', BASE + ':' + path), git('rev-parse', AFTER + ':' + path)))
reports = {}
for side in ('base', 'after'):
    h = B / side / 'hosted'
    imported = (h / 'imported.txt').read_text().splitlines()
    run_txt = (h / 'run.txt').read_text()
    lines.append('## %s (%s)' % (side, [l for l in imported if l.startswith('workflow run:')][0]))
    lines.append('run.txt: ' + '; '.join(l for l in run_txt.splitlines() if l.split(':')[0] in ('dispatched ref', 'tested commit', 'tested tree', 'inputs', 'allowance', 'partition')))
    for form in FORMS:
        shards = sorted((h / 'shards' / form).glob('shard-*.log.gz'))
        merge = (h / (form + '.merge.log')).read_text()
        m = re.search(r'merged (\d+) test262 shard report', merge)
        run = (h / (form + '.run.log')).read_text()
        ec = re.findall(r'^exit code: (\d+)', run, re.M)
        j = json.load(open(h / (form + '.test262.json')))
        binl = (h / (form + '.binaries.log')).read_text()
        dl = re.search(r'distinct lists: (\d+)', binl)
        lines.append('%s: shard logs %d of %d; merge took %s shard reports; run log exit code %s; retention retainable=%s; coverage whole=%s; every variant accounted=%s; distinct assembly lists %s' % (
            form, len(shards), j['partition']['shardCount'], m.group(1) if m else '?', ec[-1] if ec else '?', j['retention']['retainable'],
            j['coverage']['whole'], j['totals']['accountsForEveryVariant'], dl.group(1) if dl else '?'))
        rows = {}
        for l in gzip.open(h / (form + '.report.gz'), 'rt'):
            if l.startswith('result|'):
                c = l.rstrip('\n').split('|')
                rows[(c[1], c[2])] = c
        reports[(side, form)] = rows
cmp = {}
for form in FORMS:
    t = (OUT / ('compare-%s.log' % form)).read_text()
    cmp[form] = t
    m = re.search(r'(\d+) variant\(s\) present in only one report were ignored', t)
    lines.append('the comparison %s: variants present in only one report, ignored by it: %s; its exit: %s' % (form, m.group(1) if m else 'none stated', re.findall(r'^exit=(\d+)', t, re.M)))
(OUT / 'completeness.txt').write_text('\n'.join(lines) + '\n')


def verdict(c):
    return '%s/%s' % (c[3], c[5]) if c[3] == 'Exhausted' and c[5] else c[3]


rer = collections.defaultdict(list)
if RERUN:
    for form in FORMS:
        p = RERUN / ('rerun-%s.tsv' % form)
        if p.exists():
            for l in open(p):
                side, f2, rep, res = l.rstrip('\n').split('\t', 3)
                c = res.split('|')
                rer[(form, c[1], c[2], side)].append(verdict(c))
race = {}
if RACE:
    for form in FORMS:
        p = RACE / ('race-%s.tsv' % form)
        if p.exists():
            for l in open(p):
                side, f2, rep, res = l.rstrip('\n').split('\t', 3)
                c = res.split('|')
                race[(form, c[1], c[2], side)] = verdict(c)

out_rows = []
for form in FORMS:
    cls = None
    for l in cmp[form].splitlines():
        h = re.match(r"## \((.+?)\)", l)
        if h:
            cls = h.group(1)
            continue
        m = re.match(r"  (test/\S+) \[(\w+)\]: (\S+) -> (\S+)", l)
        if not m:
            continue
        path, var, b, a = m.groups()
        if cls.startswith('unclassified'):
            if b == 'Failed' and a == 'Passed':
                kind, reading = 'C1 - to be read', 'fails at the base and passes after; JSD-0084 names no correction, so a finding if admitted'
            else:
                kind, reading = 'REGRESSION', 'a difference in no class the rule admits'
            out_rows.append((form, path, var, b, a, 'unclassified', kind, '-', '-', reading))
        elif cls.startswith("c'"):
            rb, ra = rer.get((form, path, var, 'base'), []), rer.get((form, path, var, 'after'), [])
            if not rb or not ra:
                kind, reading = "C4 (c') - re-runs not yet taken", '-'
            elif len(set(rb + ra)) == 1:
                kind, reading = "C4 (c')", 're-runs agree'
            else:
                rc = (race.get((form, path, var, 'base')), race.get((form, path, var, 'after')))
                if rc[0] is None or rc[1] is None:
                    kind, reading = 'C4 (e) - race not yet taken', 're-runs differ'
                else:
                    okr = rc[0] == rc[1] and not rc[0].endswith('WallClock') and 'Defect' not in rc[0]
                    kind = 'C4 (e)' if okr else 'REGRESSION'
                    reading = 're-runs differ; race at ten times the wall clock: base %s, after %s' % rc
            out_rows.append((form, path, var, b, a, "(c')", kind, ', '.join(rb) or '-', ', '.join(ra) or '-', reading))
        elif cls[:1] in ('a', 'b') and (len(cls) == 1 or not cls[1].isalpha()):
            ok = form == 'wide-native'
            out_rows.append((form, path, var, b, a, '(%s)' % cls[0], 'C4 (%s)' % cls[0] if ok else 'REGRESSION', '-', '-', 'the comparison\'s own class, both runs native' if ok else 'a native class in the bytecode pair'))
        else:
            out_rows.append((form, path, var, b, a, cls, 'REGRESSION', '-', '-', 'a class the rule does not admit'))
    rb, ra = reports[('base', form)], reports[('after', form)]
    for k in sorted(k for k in rb if k not in ra):
        out_rows.append((form, k[0], k[1], verdict(rb[k]), '(absent)', 'in the base report only', 'REGRESSION', '-', '-', 'a row the after report does not hold'))
    for k in sorted(k for k in ra if k not in rb):
        out_rows.append((form, k[0], k[1], '(absent)', verdict(ra[k]), 'in the after report only', 'REGRESSION', '-', '-', 'a row the base report does not hold'))

with open(OUT / 'changed-verdicts.tsv', 'w', newline='\n') as o:
    o.write('# part A, SJL-17.6: every variant whose verdict or exhausted dimension differs between the base and the after run, by form, with its class: the comparison\'s rows, and the rows in one report only, which it ignores (part-a-tables.py)\n')
    o.write('form\tpath\tvariant\tbase (hosted)\tafter (hosted)\tcomparison\'s class\tclass and kind\tre-runs at the base\tre-runs at the after commit\treading\n')
    for r in out_rows:
        o.write('\t'.join(r) + '\n')

with open(OUT / 'rows-collation.txt', 'w', newline='\n') as o:
    o.write('# part A, SJL-17.6: the variants of the rule\'s named collation directories (%s), per form: how many each report holds, by verdict, and every one whose verdict or exhausted dimension differs (part-a-tables.py)\n' % ', '.join(COLLATION))
    for form in FORMS:
        rb, ra = reports[('base', form)], reports[('after', form)]
        for pre in COLLATION:
            kb = sorted(k for k in rb if k[0].startswith(pre)); ka = sorted(k for k in ra if k[0].startswith(pre))
            vb = collections.Counter(verdict(rb[k]) for k in kb); va = collections.Counter(verdict(ra[k]) for k in ka)
            diff = [k for k in sorted(set(kb) | set(ka)) if (verdict(rb[k]) if k in rb else '(absent)') != (verdict(ra[k]) if k in ra else '(absent)')]
            o.write('%s\t%s\tbase %d variants %s\tafter %d variants %s\tthe same variants: %s\tdiffering: %d\n' % (
                form, pre, len(kb), dict(sorted(vb.items())), len(ka), dict(sorted(va.items())), kb == ka, len(diff)))
            for k in diff:
                o.write('%s\t  %s [%s]\t%s -> %s\n' % (form, k[0], k[1], verdict(rb[k]) if k in rb else '(absent)', verdict(ra[k]) if k in ra else '(absent)'))
cnt = collections.Counter((r[0], r[6]) for r in out_rows)
print(open(OUT / 'completeness.txt').read())
for k, v in sorted(cnt.items()):
    print('%s\t%s\t%d' % (k[0], k[1], v))
print(open(OUT / 'rows-collation.txt').read())
