#!/usr/bin/env bash
# parts-bc.sh <base|after>: part B's slice compiler --checks --verbose, then part C's twins, each behind the lock and
# the gate, in the tree whose Release build build-roots.sh left (bundle sjl-9-001's parts-bc.sh, with this bundle's
# paths); then datasets.py reads part B's run (Python only).
S=/tmp/claude-0/-home-user/0a6f6b61-1da7-5ce6-9020-88003835d4eb/scratchpad/sjl-rest/vm-s176
side=$1; T=/dev/shm/s176/$side; case $side in base) C=68cad8baf7403125d44d5007441d30ff07bc4edf;; after) C=983d690adac63f6d117ff2cd2d9cca61fc16e3d2;; esac
SC=$T/src/compositions/Broiler.VM.Composition.JavaScript.SliceCompiler/bin/Release/net10.0
mkdir -p $S/$side/partb $S/$side/partc $S/$side/datasets
sha256sum $SC/Broiler.VM.Composition.JavaScript.SliceCompiler.dll > $S/$side/partb/slicecompiler.sha256
$S/scripts/run.sh checks-$side $S/$side/partb $T -- $SC/Broiler.VM.Composition.JavaScript.SliceCompiler --checks --verbose
started=$(awk -F'\t' '{print $4}' $S/$side/partb/runs.tsv | tail -1 | sed 's/started=//')
python3 -I $S/scripts/datasets.py $T $C $side $S/base/datasets/named-check-names.txt $S/$side/partb/checks-$side.out.txt $S/$side/partb/checks-$side.exit.txt $S/$side/partb/slicecompiler.sha256 "$started" $S/$side/datasets > $S/$side/datasets/datasets-summary.txt 2>&1
if [ "$side" = base ]; then
  $S/scripts/run.sh twins-$side $S/$side/partc $T -- python3 -I $S/scripts/twins.py base $T $S/$side/partc/fuel-parity.log $S/$side/partc/attempts.tsv
else
  $S/scripts/run.sh twins-$side $S/$side/partc $T -- python3 -I $S/scripts/twins.py after $T $S/base/partc/fuel-parity.log $S/$side/partc/fuel-parity.log $S/$side/partc/attempts.tsv
fi
echo DONE > $S/$side/parts-bc.DONE
